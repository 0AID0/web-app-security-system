JWT (JSON Web Tokens)

Интерактивное руководство по безопасной аутентификации

Основы JWT

RFC 7519

JSON Web Token (JWT) — это открытый стандарт для создания токенов доступа, основанный на формате JSON. Обеспечивает безопасную передачу информации между сторонами в виде JSON-объекта.

Структура JWT:

Header {"alg": "HS256", "typ": "JWT"} - алгоритм подписи и тип токена
Payload {"sub": "123", "name": "John", "iat": 1516239022} - данные пользователя и метаданные
Signature HMACSHA256(base64UrlEncode(header) + "." + base64UrlEncode(payload), secret) - подпись для верификации

 Безопасность JWT

                
                

                    
                    

                        Основные угрозы:
                        
    
                            
  • XSS атаки - если токен хранится в localStorage
    • 
                            
  • CSRF атаки - если токен хранится в cookies
    • 
                            
  • Перехват токена - при использовании HTTP вместо HTTPS
    • 
                            
  • Подмена алгоритма - изменение "alg": "none"
    • 
                            
  • Утечка секретного ключа - компрометация сервера
    • 
                        

                    

                

                
                
Рекомендации по безопасности:

                

                    
                    

                        
    
                            
  • Всегда используйте HTTPS для защиты от перехвата
    • 
                            
  • Храните секретные ключи в env-переменных на сервере
    • 
                            
  • Используйте короткоживущие access tokens (15-30 мин) и долгоживущие refresh tokens
    • 
                            
  • Реализуйте отзыв токенов через blacklist
    • 
                            
  • Проверяйте подпись каждого токена
    • 
                            
  • Используйте соответствующий алгоритм (HS256 для монолита, RS256 для микросервисов)
    • 
                            
  • Ограничивайте данные в payload - не храните чувствительную информацию
    •

Генерация и проверка токенов

                        
const jwt = require('jsonwebtoken');
const express = require('express');
const app = express();
require('dotenv').config();

// Секретные ключи (должны быть в .env)
const ACCESS_TOKEN_SECRET = process.env.ACCESS_TOKEN_SECRET;
const REFRESH_TOKEN_SECRET = process.env.REFRESH_TOKEN_SECRET;

// Генерация токенов
app.post('/login', (req, res) => {
    const { username, password } = req.body;
    
    // Проверка учетных данных (в реальном приложении - проверка в БД)
    const user = { id: 123, username, role: 'user' };
    
    // Access Token (короткоживущий)
    const accessToken = jwt.sign(
        { userId: user.id, role: user.role },
        ACCESS_TOKEN_SECRET,
        { expiresIn: '15m' }
    );
    
    // Refresh Token (долгоживущий)
    const refreshToken = jwt.sign(
        { userId: user.id },
        REFRESH_TOKEN_SECRET,
        { expiresIn: '7d' }
    );
    
    // Отправляем access token в ответе, refresh token в httpOnly cookie
    res.cookie('refreshToken', refreshToken, {
        httpOnly: true,
        secure: true,
        sameSite: 'strict',
        maxAge: 7 * 24 * 60 * 60 * 1000 // 7 дней
    }).json({ accessToken });
});

// Обновление токена
app.post('/refresh', (req, res) => {
    const refreshToken = req.cookies.refreshToken;
    if (!refreshToken) return res.sendStatus(401);
    
    jwt.verify(refreshToken, REFRESH_TOKEN_SECRET, (err, user) => {
        if (err) return res.sendStatus(403);
        
        const newAccessToken = jwt.sign(
            { userId: user.userId },
            ACCESS_TOKEN_SECRET,
            { expiresIn: '15m' }
        );
        
        res.json({ accessToken: newAccessToken });
    });
});

                            const getToken = () => {
                                return localStorage.getItem('accessToken');
                            };
                            
                            // Выход из системы
                            const logout = () => {
                                removeToken();
                                // Дополнительные действия при выходе (редирект, очистка состояния и т.д.)
                                window.location.href = '/login';
                            };
                            
                            // Добавление токена в заголовки и обработка 401 ошибки
                            const fetchWithAuth = async (url, options = {}) => {
                                const token = getToken();
                                if (!token) throw new Error('No token found');
                                
                                const headers = {
                                    'Content-Type': 'application/json',
                                    'Authorization': `Bearer ${token}`,
                                    ...options.headers
                                };
                                
                                const response = await fetch(url, { 
                                    ...options, 
                                    headers,
                                    credentials: 'include' // Для работы с httpOnly cookies
                                });
                                
                                // Если токен истек, пробуем обновить
                                if (response.status === 401) {
                                    const newToken = await refreshToken();
                                    if (newToken) {
                                        // Повторяем запрос с новым токеном
                                        headers.Authorization = `Bearer ${newToken}`;
                                        return fetch(url, { ...options, headers });
                                    }
                                    // Если не удалось обновить - разлогиниваем
                                    logout();
                                    throw new Error('Session expired. Please login again.');
                                }
                                
                                // Если другая ошибка
                                if (!response.ok) {
                                    throw new Error(`HTTP error! status: ${response.status}`);
                                }
                                
                                return response;
                            };
                            
                            // Обновление токена
                            const refreshToken = async () => {
                                try {
                                    const response = await fetch('/api/refresh', {
                                        method: 'POST',
                                        credentials: 'include' // Для отправки httpOnly cookie с refresh token
                                    });
                                    
                                    if (response.ok) {
                                        const { accessToken } = await response.json();
                                        storeToken(accessToken);
                                        return accessToken;
                                    }
                                    
                                    // Если refresh не удался
                                    return null;
                                } catch (error) {
                                    console.error('Refresh token failed:', error);
                                    return null;
                                }
                            };

                                    return data;

                                try {
                                    const payload = JSON.parse(atob(token.split('.')[1]));
                                    if (payload.exp && payload.exp < Date.now() / 1000) {

                                    }
                                    return false;

                                    return true;
                                }

                                try {
                                    return JSON.parse(atob(token.split('.')[1]));
                                } catch {
                                    return null;

                                const token = getToken();

                                    // Пробуем обновить токен при загрузке приложения
                                    refreshToken().catch(() => logout());
                                }
                            };
                            
                            // Вызываем инициализацию при загрузке
                            initAuth();
                        

                        
                    

                    
                    

                        
Middleware для проверки JWT

                        
// Middleware для проверки JWT
const authenticateJWT = (req, res, next) => {
    const authHeader = req.headers['authorization'];
    const token = authHeader && authHeader.split(' ')[1];
    
    if (!token) return res.sendStatus(401);
    
    jwt.verify(token, ACCESS_TOKEN_SECRET, (err, user) => {
        if (err) return res.sendStatus(403);
        req.user = user;
        next();
    });
};

// Middleware для проверки ролей
const checkRole = (role) => {
    return (req, res, next) => {
        if (req.user.role !== role) {
            return res.sendStatus(403);
        }
        next();
    };
};

// Защищенные маршруты
app.get('/protected', authenticateJWT, (req, res) => {
    res.json({ message: `Hello user ${req.user.userId}` });
});

app.get('/admin', authenticateJWT, checkRole('admin'), (req, res) => {
    res.json({ message: 'Admin dashboard' });
});

// Обработка ошибок
app.use((err, req, res, next) => {
    console.error(err.stack);
    res.status(500).json({ error: 'Internal Server Error' });
});
