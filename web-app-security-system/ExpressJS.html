Обзор технологии

Express.js — это минималистичный и гибкий веб-фреймворк для Node.js, предоставляющий robust набор функций для веб- и мобильных приложений с мощной системой middleware для обеспечения безопасности.

Безопасность

Встроенная поддержка middleware для защиты от CSRF, XSS, инъекций и других атак.

Производительность

Высокая скорость работы благодаря асинхронной природе Node.js.

Расширяемость

Богатая экосистема middleware и плагинов для любых задач.

Использование Middleware

const express = require('express');
const app = express();

// Middleware для логирования
app.use((req, res, next) => {
  console.log(`${new Date().toISOString()} - ${req.method} ${req.url}`);
  next();
});

// Middleware для обработки JSON
app.use(express.json());

// Middleware для статических файлов
app.use(express.static('public'));

// Обработка POST запроса
app.post('/api/data', (req, res) => {
  const data = req.body;
  res.json({ received: data });
});

app.listen(3000);

Популярные Middleware
Популярные middleware для Express:
- body-parser: Парсинг тела запроса
- cookie-parser: Работа с cookies
- compression: Сжатие ответов
- cors: Включение CORS
- helmet: Безопасность HTTP-заголовков
- morgan: Логирование запросов
- express-session: Управление сессиями
- csurf: Защита от CSRF

Порядок middleware важен! Они выполняются в порядке их объявления.

Безопасность в Express

const express = require('express');
const helmet = require('helmet');
const rateLimit = require('express-rate-limit');
const csrf = require('csurf');
const cookieParser = require('cookie-parser');

const app = express();

// Безопасные HTTP-заголовки
app.use(helmet());

// Ограничение запросов
app.use(rateLimit({
  windowMs: 15 * 60 * 1000, // 15 минут
  max: 100 // максимум 100 запросов с одного IP
}));

// Защита от CSRF
app.use(cookieParser());
app.use(csrf({ cookie: true }));

// Обработка POST с CSRF-токеном
app.post('/secure', (req, res) => {
  if (!req.csrfToken()) {
    return res.status(403).send('Неверный CSRF-токен');
  }
  res.send('Защищенный запрос выполнен');
});

Проверка безопасности

Рекомендации по безопасности
Основные рекомендации по безопасности:
1. Всегда используйте helmet для безопасных заголовков
2. Ограничивайте запросы с помощью express-rate-limit
3. Используйте HTTPS с помощью express-enforces-ssl
4. Проверяйте входные данные (санитизация)
5. Используйте CSRF-токены для форм
6. Храните секреты в переменных окружения
7. Обновляйте зависимости (npm audit)
8. Используйте ORM/ODM для защиты от инъекций
9. Настройте CORS только для нужных доменов
10. Используйте куки с флагами Secure и HttpOnly

Создание REST API

const express = require('express');
const app = express();
app.use(express.json());

// "База данных"
let products = [
  { id: 1, name: 'Ноутбук', price: 50000 },
  { id: 2, name: 'Смартфон', price: 30000 }
];

// GET /api/products
app.get('/api/products', (req, res) => {
  res.json(products);
});

// GET /api/products/:id
app.get('/api/products/:id', (req, res) => {
  const product = products.find(p => p.id === parseInt(req.params.id));
  if (!product) return res.status(404).send('Продукт не найден');
  res.json(product);
});

// POST /api/products
app.post('/api/products', (req, res) => {
  const product = {
    id: products.length + 1,
    name: req.body.name,
    price: req.body.price
  };
  products.push(product);
  res.status(201).json(product);
});

// PUT /api/products/:id
app.put('/api/products/:id', (req, res) => {
  const product = products.find(p => p.id === parseInt(req.params.id));
  if (!product) return res.status(404).send('Продукт не найден');
  product.name = req.body.name;
  product.price = req.body.price;
  res.json(product);
});

// DELETE /api/products/:id
app.delete('/api/products/:id', (req, res) => {
  products = products.filter(p => p.id !== parseInt(req.params.id));
  res.status(204).send();
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => console.log(`Сервер запущен на порту ${PORT}`));

Аутентификация в Express

const express = require('express');
const bcrypt = require('bcrypt');
const jwt = require('jsonwebtoken');
const app = express();

app.use(express.json());

// "База данных" пользователей
const users = [];

// Регистрация
app.post('/api/register', async (req, res) => {
  try {
    const hashedPassword = await bcrypt.hash(req.body.password, 10);
    const user = {
      id: Date.now().toString(),
      name: req.body.name,
      email: req.body.email,
      password: hashedPassword
    };
    users.push(user);
    res.status(201).send();
  } catch {
    res.status(500).send();
  }
});

// Вход
app.post('/api/login', async (req, res) => {
  const user = users.find(u => u.email === req.body.email);
  if (!user) return res.status(400).send('Пользователь не найден');
  try {
    if (await bcrypt.compare(req.body.password, user.password)) {
      const accessToken = jwt.sign(
        { id: user.id, name: user.name },
        process.env.ACCESS_TOKEN_SECRET,
        { expiresIn: '15m' }
      );
      res.json({ accessToken });
    } else {
      res.send('Неверный пароль');
    }
  } catch {
    res.status(500).send();
  }
});

// Защищенный маршрут
app.get('/api/profile', authenticateToken, (req, res) => {
  res.json(users.find(u => u.id === req.user.id));
});

function authenticateToken(req, res, next) {
  const authHeader = req.headers['authorization'];
  const token = authHeader && authHeader.split(' ')[1];
  if (!token) return res.sendStatus(401);
  jwt.verify(token, process.env.ACCESS_TOKEN_SECRET, (err, user) => {
    if (err) return res.sendStatus(403);
    req.user = user;
    next();
  });
}

app.listen(3000);

WebSocket с Express

const express = require('express');
const http = require('http');
const WebSocket = require('ws');

const app = express();
const server = http.createServer(app);
const wss = new WebSocket.Server({ server });

// HTTP маршрут
app.get('/', (req, res) => {
  res.sendFile('index.html', { root: __dirname });
});

// WebSocket соединения
wss.on('connection', (ws) => {
  console.log('Новое соединение');
  ws.on('message', (message) => {
    console.log('Получено: ', message);
    // Отправить сообщение всем клиентам
    wss.clients.forEach(client => {
      if (client.readyState === WebSocket.OPEN) {
        client.send(message);
      }
    });
  });
  ws.send('Добро пожаловать в чат!');
});

server.listen(3000, () => {
  console.log('Сервер запущен на порту 3000');
});
